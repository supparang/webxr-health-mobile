function evaluateSubmissionWithRubric(stageCode, isBoss, labText, refText) {
  const kws = STAGE_KEYWORDS[stageCode] || [];
  const combined = (String(labText) + " " + String(refText)).toLowerCase();
  const matched = kws.filter(k => combined.indexOf(k.toLowerCase()) !== -1);

  let labLevel = 1;
  if (labText.length >= 90 && matched.length >= 3) labLevel = 4;
  else if (labText.length >= 60 && matched.length >= 2) labLevel = 3;
  else if (labText.length >= 30 && matched.length >= 1) labLevel = 2;

  let refLevel = 1;
  if (refText.length >= 80 && matched.length >= 2) refLevel = 4;
  else if (refText.length >= 50 && matched.length >= 1) refLevel = 3;
  else if (refText.length >= 25) refLevel = 2;

  const labMax = isBoss ? 150 : 60;
  const refMax = isBoss ? 100 : 40;
  return {
    labLevel: labLevel,
    refLevel: refLevel,
    labScore: Math.round(labMax * (labLevel / 4)),
    refScore: Math.round(refMax * (refLevel / 4)),
    matchedKeywords: matched.join(", ")
  };
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let pSheet = ss.getSheetByName("Players") || ss.insertSheet("Players");
    let sSheet = ss.getSheetByName("Submissions") || ss.insertSheet("Submissions");

    if (sSheet.getLastRow() === 0) {
      sSheet.appendRow([
        "Timestamp", "PlayerName", "StageCode", "QuizScore",
        "LabScore", "LabLevel", "RefScore", "RefLevel",
        "TotalStageScore", "MatchedKeywords", "LabText", "ReflectionText"
      ]);
    }

    const grade = evaluateSubmissionWithRubric(
      data.stageCode, Boolean(data.isBoss), data.labOutput || "", data.reflectionText || ""
    );

    const quizScore = Number(data.quizScore || 0);
    const totalStageScore = quizScore + grade.labScore + grade.refScore;

    sSheet.appendRow([
      new Date(), data.playerName, data.stageCode, quizScore,
      grade.labScore, "Level " + grade.labLevel,
      grade.refScore, "Level " + grade.refLevel,
      totalStageScore, grade.matchedKeywords,
      data.labOutput, data.reflectionText
    ]);

    pSheet.appendRow([new Date(), data.playerName, data.unlockedStage, data.totalScore, data.stageCode]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", grade: grade }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
